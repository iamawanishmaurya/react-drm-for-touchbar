```ts
import { execFileSync, type ExecFileSyncOptions } from 'child_process';
import fs from 'fs';
import path from 'path';

// Upper bound for a synchronous helper command. The bar's render loop runs on
// the same thread, so a hung helper (e.g. wpctl while WirePlumber is stuck)
// would freeze the whole bar.
export const EXEC_TIMEOUT_MS = Number(process.env.REACT_DRM_EXEC_TIMEOUT_MS) || 1500;

// execFileSync with a hard timeout (SIGKILL) and stderr discarded. Throws on
// timeout, non-zero exit or a missing binary — callers fall back.
export function execQuick(cmd: string, args: string[], opts: ExecFileSyncOptions = {}): string {
  return execFileSync(cmd, args, {
    timeout: EXEC_TIMEOUT_MS,
    killSignal: 'SIGKILL',
    stdio: ['ignore', 'pipe', 'ignore'],
    ...opts,
    encoding: 'utf8',
  }) as string;
}

// True when `cmd` is an executable on PATH — checked without running it, so a
// hung daemon at startup can't make us pick the wrong backend for good.
export function commandExists(cmd: string, envPath = process.env.PATH ?? ''): boolean {
  return envPath.split(path.delimiter).some(dir => {
    if (!dir) return false;
    try { fs.accessSync(path.join(dir, cmd), fs.constants.X_OK); return true; }
    catch { return false; }
  });
}
```

Why each symbol exists, one line each:

- `EXEC_TIMEOUT_MS` — the single number every exec call in the app shares; read once at
  module load from `REACT_DRM_EXEC_TIMEOUT_MS` so a slow machine can be tuned without a
  code change, defaulting to 1500 ms.
- `execQuick()` — `execFileSync` with `timeout` + `killSignal: 'SIGKILL'`, so a hung child
  is killed rather than blocking the bar's only thread; stderr is discarded because
  helper chatter is never actionable. It throws, and every caller already wraps in
  try/catch and falls back.
- `commandExists()` — replaces `execFileSync('wpctl', ['--help'])` backend detection with
  a pure `fs.accessSync(X_OK)` scan of `PATH`, so detection can never itself hang and a
  transient failure cannot permanently pin the wrong backend (FREEZE-03).

Do NOT export anything else from this file, and do NOT add options to `execQuick`. Note
for the report only: because `...opts` is spread after `timeout`, a caller could override
the timeout; no caller does today.
