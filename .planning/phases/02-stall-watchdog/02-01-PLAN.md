### Three findings the planner made that change how this phase must be executed

**1. The heartbeat must be seeded synchronously, not just on the first interval tick.**
`SharedArrayBuffer` is zero-initialised. A beat that has never been written reads as
"last heartbeat at time 0", which is older than any limit — the worker would `SIGKILL` a
perfectly healthy bar one second into every start. `startStallWatchdog` therefore stores
one beat before it arms the interval. This is an implementation detail the phase file
implies ("every 1000 ms, `Atomics.store`") but does not spell out; without it every test
except `pause` becomes non-deterministic.

**2. `ℹ tests 1` does NOT prove a `--test-name-pattern` matched — the FILE gets counted.**
Measured on node 26 in this repo: a correctly anchored pattern prints
`✔ <test name> (2.16ms)` / `ℹ tests 1` / `ℹ pass 1`, but a pattern matching **nothing**
prints `✔ tests/sessionRuntime.test.ts (178ms)` / `ℹ tests 1` / `ℹ pass 1` and exits 0 —
the whole file counts as one passing "test". So `ℹ tests 1` plus exit 0 prove nothing. The
only signal that separates the two is the ✔/✖ **line text**: a matched test prints its own
title, an unmatched one prints the file path. Task 4's gate therefore asserts
`green_named` (the title appears) and `green_fileline` / `red_fileline` (the path line is
absent). Measured formats: `✔ NAME (1.51ms)` appears once; `✖ NAME (2.37ms)` appears
**twice** (once in the run, once under the `✖ failing tests:` summary), so the red
assertion is "at least one", never "exactly one".

**3. `stallMs=1000` against the default `beatMs=1000` makes the `short` test a coin flip.**
The worst stale beat is `blockMs + beatMs`. The design default gives a 15 s limit against a
1 s heartbeat — a 14 s margin, which is the point. But the suite drives the limit down to
1000 ms, where the margin is **zero** and the sub-millisecond difference between two
independent 1000 ms timers (one on the main thread, one in the worker) decides whether
`short` passes. The fixture therefore passes `beatMs: 200`: worst case 600 + 200 = 800 ms,
a 200 ms margin under the 1000 ms limit. The bar itself keeps the default 1000 ms.

```ts
import { Worker } from 'worker_threads';

// The polling loop lives in the worker as plain JS text rather than a second
// module: the bar runs from source under `tsx` and from compiled `dist/` once
// Phase 3 switches the unit over, and an eval'd string resolves in both without
// a build step or a path that only exists in one of the two layouts.
const WORKER_SOURCE = `
const fs = require('node:fs');
const { workerData } = require('node:worker_threads');
const beat = new BigInt64Array(workerData.sab);
const stallMs = workerData.stallMs;
// -1 is the pause sentinel written by pause(); a healthy heartbeat is a
// CLOCK_MONOTONIC millisecond stamp and can never be negative.
const PAUSED = -1n;
setInterval(() => {
  const now = process.hrtime.bigint() / 1000000n;
  const last = Atomics.load(beat, 0);
  if (last === PAUSED) return;
  const stalled = now - last;
  if (stalled <= stallMs) return;
  fs.writeSync(2, '[stall-watchdog] main thread blocked for ' + stalled + ' ms (limit ' + stallMs + ') \\u2014 exiting so systemd restarts the bar\\n');
  process.kill(process.pid, 'SIGKILL');
}, 1000);
`;

export interface StallWatchdog {
  pause(): void;
  resume(): void;
  stop(): void;
}

// CLOCK_MONOTONIC, not Date.now(): it does not advance while the machine is
// suspended, so waking from suspend can never read as a multi-hour stall. A
// SIGTERM handler could not do this job either — a blocked main thread runs no
// handlers, which is why this has to be a second thread, and why the kill is
// SIGKILL rather than something polite: systemd's Restart=always is what turns
// the death back into a working bar, and a blocked thread cannot cooperate
// with a graceful shutdown either.
export function startStallWatchdog(opts: { stallMs: number; beatMs?: number }): StallWatchdog {
  const { stallMs } = opts;
  const beatMs = opts.beatMs ?? 1000;

  const sab = new SharedArrayBuffer(8);
  const beat = new BigInt64Array(sab);
  const PAUSED = -1n;

  const nowMs = (): bigint => process.hrtime.bigint() / 1_000_000n;

  // Seed before arming the interval: the buffer starts as all zeroes, and a
  // zero beat reads as "last seen at time 0", which is instantly past the
  // limit. Without this the worker kills a healthy bar one second into startup.
  Atomics.store(beat, 0, nowMs());

  const timer = setInterval(() => Atomics.store(beat, 0, nowMs()), beatMs);
  timer.unref();

  const worker = new Worker(WORKER_SOURCE, { eval: true, workerData: { sab, stallMs } });
  worker.unref();

  return {
    pause: () => { Atomics.store(beat, 0, PAUSED); },
    // Fresh beat on the way back: the first tick after a resume would otherwise
    // compare against the pre-suspend stamp and kill the bar.
    resume: () => { Atomics.store(beat, 0, nowMs()); },
    stop: () => {
      clearInterval(timer);
      void worker.terminate();
    },
  };
}
```

Why each decision exists, one line each:

- `WORKER_SOURCE` as a string with `eval: true` — identical behaviour under `tsx` (source)
  and `dist/` (compiled), which a `new Worker(new URL('./watchdog.ts', import.meta.url))`
  could not give us across both.
- `Atomics.store` / `Atomics.load` on a `BigInt64Array` — the only atomic RMW the
  `Atomics` API offers over an 8-byte shared cell, and it is atomic across threads without
  a lock.
- `process.hrtime.bigint() / 1_000_000n` — `CLOCK_MONOTONIC` in milliseconds, immune to
  suspend and to wall-clock jumps.
- `-1n` pause sentinel — a single value the worker can recognise without a second shared
  cell or a lock; every real stamp is non-negative.
- `timer.unref()` and `worker.unref()` — neither handle may keep the process alive. This
  is what the `idle` test proves.
- `fs.writeSync(2, …)` — synchronous and unbuffered, so the reason lands in
  `journalctl -u react-drm` even though the next instruction kills the process.
- `process.kill(process.pid, 'SIGKILL')` — uncatchable, so nothing in the blocked main
  thread can veto or delay it.

The `\\u2014` in the template literal emits a real em dash in the worker's source; using the
literal `—` character would also be fine, but the escape keeps the string pure ASCII in the
source file. The `\\n` is a two-character escape **inside the worker source string** — i.e.
the TS file literally contains `\\n`, which becomes `\n` in the text the worker evaluates.

Do not export anything beyond `StallWatchdog` and `startStallWatchdog`, and do not add a
default value for `stallMs` here — `index.tsx` owns the `REACT_DRM_STALL_MS` default and the
`0`-disables rule.

```ts
// Child process for tests/stallWatchdog.test.ts. Spawned by that suite, never
// imported by it. Relative imports only: this runs under `--import tsx` with no
// tsconfig of its own, so the `@/*` alias would not resolve.
import { startStallWatchdog } from '../../lib/services/stallWatchdog';

const stallMs = Number(process.argv[2]);
const mode = process.argv[3];

// 200 ms heartbeat, not the 1000 ms default: the suite drives the limit down to
// 1000 ms, where a 1000 ms heartbeat leaves a zero margin and the jitter between
// two independent 1000 ms timers decides the `short` case. Worst stale beat is
// then 600 + 200 = 800 ms, a 200 ms margin under the limit. The bar itself keeps
// the default.
const watchdog = startStallWatchdog({ stallMs, beatMs: 200 });

// A busy loop, not a blocking syscall: this is what a stuck render loop or a
// deadlocked native binding looks like from the outside — the thread never
// yields and never reaches another timer callback.
function block(ms: number): void {
  const end = performance.now() + ms;
  while (performance.now() < end) { /* pretend this is the bar's render loop */ }
}

switch (mode) {
  case 'block':
    // No way back out: only the watchdog can end this child. The write happens
    // 200 ms in so the heartbeat is definitely fresh when the block starts.
    setTimeout(() => { for (;;) { /* blocked forever */ } }, 200);
    break;
  case 'short':
    // A hitch under the limit must NOT kill the bar — this is the false-positive
    // guard for the 15 s production number.
    setTimeout(() => block(600), 200);
    // Holds the event loop open so the heartbeat keeps ticking. Empty on
    // purpose, and no process.exit anywhere: when this fires and nothing else is
    // ref'd, node exits by itself with 0 — which is the point of `idle`.
    setTimeout(() => {}, 2500);
    break;
  case 'idle':
    setTimeout(() => {}, 2500);
    break;
  case 'paused':
    // Must survive a stall LONGER than the limit, or every suspend/resume would
    // read as a freeze and kill the bar.
    watchdog.pause();
    block(2500);
    watchdog.stop();
    break;
  default:
    throw new Error(`unknown mode: ${mode}`);
}
```

Nothing in this file calls `process.exit` — that is what makes the `idle` and `paused`
results meaningful.

export const BLOCK_TEST_RE = `^${BLOCK_TEST_NAME}$`;

function run(stallMs: number, mode: string, timeout: number) {
  return spawnSync(process.execPath, ['--import', 'tsx', FIXTURE, String(stallMs), mode], {
    encoding: 'utf8',
    timeout,
  });

test(BLOCK_TEST_NAME, () => {
  const t0 = performance.now();
  const r = run(1000, 'block', 6000);
  const ms = performance.now() - t0;
  // Asserted FIRST and on its own: in the red-before-green run the child is
  // never killed, spawnSync times out at 6 s and SIGTERMs it, so this must be
  // the assertion that reports "the watchdog did not fire" rather than the
  // spawn error underneath it.
  assert.equal(r.signal, 'SIGKILL',
    `child was NOT killed by the watchdog — signal=${r.signal} status=${r.status} spawnError=${r.error && r.error.code} after ${ms.toFixed(0)}ms`);
  assert.equal(r.error, undefined, `spawnSync error: ${r.error && r.error.code}`);
  assert.ok(ms < 6000, `the watchdog took ${ms.toFixed(0)}ms to notice a blocked main thread`);
  assert.match(r.stderr, /\[stall-watchdog\] main thread blocked for \d+ ms \(limit 1000\)/, r.stderr);
});

  const r = run(1000, 'short', 15000);
  assert.equal(r.signal, null, `child was killed by ${r.signal}: ${r.stderr}`);
  assert.equal(r.status, 0, `child exited ${r.status}: ${r.stderr}`);
  assert.doesNotMatch(r.stderr, /\[stall-watchdog\]/, r.stderr);
});

test('an idle bar exits on its own — the watchdog holds nothing open', () => {
  const r = run(1000, 'idle', 15000);
  assert.equal(r.signal, null, `child was killed by ${r.signal}: ${r.stderr}`);
  assert.equal(r.status, 0, `child exited ${r.status}: ${r.stderr}`);
  assert.doesNotMatch(r.stderr, /\[stall-watchdog\]/, r.stderr);
});

test('a paused bar survives a stall longer than the limit', () => {
  const r = run(1000, 'paused', 15000);
  assert.equal(r.signal, null, `child was killed by ${r.signal} — pause() did not hold`);
